-- Представление 1: на основании таблиц базы данных
CREATE VIEW [dbo].[Представление_Сессии]
AS
SELECT s.СессияID,
       k.ФИО             AS Клиент,
       p.ФИО             AS Психолог,
       u.Наименование    AS Услуга,
       kp.Наименование   AS Категория,
       f.Наименование    AS Формат,
       s.ДатаВремяСессии,
       u.Стоимость,
       s.СтатусОплаты
FROM Сессии AS s
     JOIN Клиенты          AS k  ON k.КлиентID     = s.КлиентID
     JOIN Психологи        AS p  ON p.ПсихологID   = s.ПсихологID
     JOIN Услуги           AS u  ON u.УслугаID     = s.УслугаID
     JOIN КатегорииПроблем AS kp ON kp.КатегорияID = u.КатегорияID
     JOIN Форматы          AS f  ON f.ФорматID     = u.ФорматID;

GO
/****** Объект:  View [dbo].[Представление_ДолгиКлиентов]    Дата создания скрипта: 05.10.2026 23:54:33 ******/ 
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
CREATE   VIEW [dbo].[Представление_ДолгиКлиентов] AS
SELECT
  c.КлиентID,
  c.ФИО,
  COUNT(s.СессияID) AS НеоплаченныхСессий,
  SUM(COALESCE(u.Стоимость, p.СтоимостьПриема, 0)) AS СуммаДолга
FROM dbo.Сессии s
JOIN dbo.Клиенты c ON s.КлиентID = c.КлиентID
LEFT JOIN dbo.Услуги u ON s.УслугаID = u.УслугаID
LEFT JOIN dbo.Психологи p ON s.ПсихологID = p.ПсихологID
WHERE s.СтатусОплаты = 0
GROUP BY c.КлиентID, c.ФИО;

GO

-- Представление 1 (рисунок 14). Загруженность психологов
CREATE VIEW [dbo].[vw_ЗагруженностьПсихологов]
AS
SELECT p.ПсихологID,
       p.ФИО,
       sp.Наименование AS Специализация,
       COUNT(s.СессияID) AS ВсегоСессий,
       SUM(CASE WHEN s.СтатусОплаты = 1 THEN 1 ELSE 0 END) AS Оплачено,
       SUM(CASE WHEN s.СтатусОплаты = 0 THEN 1 ELSE 0 END) AS НеОплачено,
       MAX(s.ДатаВремяСессии) AS ПоследняяСессия
FROM Психологи p
     JOIN Специализации sp ON sp.СпециализацияID = p.СпециализацияID
     LEFT JOIN Сессии s ON s.ПсихологID = p.ПсихологID
GROUP BY p.ПсихологID, p.ФИО, sp.Наименование;
GO

-- Представление 2 (рисунок 15). Сводка по клиентам
CREATE VIEW [dbo].[vw_СводкаПоКлиентам]
AS
SELECT k.КлиентID,
       k.ФИО,
       COUNT(s.СессияID) AS ВсегоСессий,
       SUM(CASE WHEN s.СтатусОплаты = 0 THEN 1 ELSE 0 END) AS НеоплаченныхСессий,
       MAX(s.ДатаВремяСессии) AS ПоследняяСессия
FROM Клиенты k
     LEFT JOIN Сессии s ON s.КлиентID = k.КлиентID
GROUP BY k.КлиентID, k.ФИО;

-- Хранимая процедура (рисунок 16). Запись клиента на сессию
CREATE PROCEDURE [dbo].[usp_ДобавитьСессию]
    @ПсихологID int,
    @КлиентID int,
    @УслугаID int,
    @ДатаВремя datetime2(7)
AS
BEGIN
    SET NOCOUNT ON;
    IF EXISTS (SELECT 1 FROM dbo.Сессии WHERE ПсихологID = @ПсихологID AND ДатаВремяСессии = @ДатаВремя)
    BEGIN
        RAISERROR(N'У психолога уже есть сессия на это время.',16,1);
        RETURN;
    END
    INSERT INTO dbo.Сессии (ПсихологID, КлиентID, УслугаID, ДатаВремяСессии, СтатусОплаты)
    VALUES (@ПсихологID, @КлиентID, @УслугаID, @ДатаВремя, 0);
    SELECT CAST(SCOPE_IDENTITY() AS int) AS СессияID;
END;
GO
