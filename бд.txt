Процедуры и представления CREATE OR ALTER PROCEDURE dbo.usp_ДобавитьСессию @ПсихологID INT, @КлиентID INT, @УслугаID INT, @ДатаВремя DATETIME2(7) AS BEGIN SET NOCOUNT ON; BEGIN TRY IF EXISTS (SELECT 1 FROM dbo.Сессии WHERE ПсихологID = @ПсихологID AND ДатаВремяСессии = @ДатаВремя) THROW 50001, N'У психолога уже есть сессия на это время.', 1;
    INSERT INTO dbo.Сессии (ПсихологID, КлиентID, УслугаID, ДатаВремяСессии, СтатусОплаты)
    VALUES (@ПсихологID, @КлиентID, @УслугаID, @ДатаВремя, 0);

    SELECT CAST(SCOPE_IDENTITY() AS INT) AS СессияID;
END TRY
BEGIN CATCH
    DECLARE @msg NVARCHAR(4000) = ERROR_MESSAGE();
    RAISERROR(@msg, 16, 1);
END CATCH
END GO

-- 5. Триггеры: аудит сессий и обновление статуса по оплатам CREATE OR ALTER TRIGGER dbo.trg_Сессии_Аудит ON dbo.Сессии AFTER INSERT, UPDATE, DELETE AS BEGIN SET NOCOUNT ON;
-- INSERT: подставляем старое = вставленному, чтобы не было NULL (по вашему требованию)
INSERT INTO dbo.АудитСессий (Операция, СессияID, СтарыйСтатус, НовыйСтатус, СтараяДатаВремя, НоваяДатаВремя, Пользователь, ДатаВремяОперации)
SELECT N'INSERT', i.СессияID,
       COALESCE(i.СтатусОплаты, 0),
       COALESCE(i.СтатусОплаты, 0),
       COALESCE(i.ДатаВремяСессии, '1900-01-01'),
       i.ДатаВремяСессии,
       SUSER_SNAME(), SYSUTCDATETIME()
FROM inserted i
LEFT JOIN deleted d ON d.СессияID = i.СессияID
WHERE d.СессияID IS NULL;

-- UPDATE
INSERT INTO dbo.АудитСессий (Операция, СессияID, СтарыйСтатус, НовыйСтатус, СтараяДатаВремя, НоваяДатаВремя, Пользователь, ДатаВремяОперации)
SELECT N'UPDATE', i.СессияID,
       COALESCE(d.СтатусОплаты, i.СтатусОплаты, 0),
       COALESCE(i.СтатусОплаты, 0),
       COALESCE(d.ДатаВремяСессии, i.ДатаВремяСессии, '1900-01-01'),
       COALESCE(i.ДатаВремяСессии, '1900-01-01'),
       SUSER_SNAME(), SYSUTCDATETIME()
FROM inserted i
INNER JOIN deleted d ON d.СессияID = i.СессияID;

-- DELETE
INSERT INTO dbo.АудитСессий (Операция, СессияID, СтарыйСтатус, СтараяДатаВремя, Пользователь, ДатаВремяОперации)
SELECT N'DELETE', d.СессияID,
       COALESCE(d.СтатусОплаты, 0),
       COALESCE(d.ДатаВремяСессии, '1900-01-01'),
       SUSER_SNAME(), SYSUTCDATETIME()
FROM deleted d
LEFT JOIN inserted i ON i.СессияID = d.СессияID
WHERE i.СессияID IS NULL;

END GO
-- Триггер: после вставки оплат обновлять статус CREATE OR ALTER TRIGGER dbo.trg_Оплаты_ОбновитьСтатус ON dbo.Оплаты AFTER INSERT AS BEGIN SET NOCOUNT ON;
;WITH ChangedSessions AS (
  SELECT DISTINCT СессияID FROM inserted WHERE СессияID IS NOT NULL
)
UPDATE s
SET СтатусОплаты = CASE WHEN paid.TotalPaid >= due.DueAmount AND due.DueAmount > 0 THEN 1 ELSE 0 END
FROM dbo.Сессии s
INNER JOIN ChangedSessions ch ON s.СессияID = ch.СессияID
CROSS APPLY (SELECT ISNULL(SUM(o.Сумма),0) AS TotalPaid FROM dbo.Оплаты o WHERE o.СессияID = ch.СессияID) paid
CROSS APPLY (SELECT TOP(1) COALESCE(u.Стоимость, p.СтоимостьПриема, 0) AS DueAmount
             FROM dbo.Сессии ss
             LEFT JOIN dbo.Услуги u ON ss.УслугаID = u.УслугаID
             LEFT JOIN dbo.Психологи p ON ss.ПсихологID = p.ПсихологID
             WHERE ss.СессияID = ch.СессияID) due;
END GO
